= blAST

== Types

string:: "value"
integer:: 123
float:: 12.34 .56 78.
symbol::
- without arity (has arity 0): sym f' x?
- with arity: sym/0 f'/1 +/2 x?/1
block:: { 1 } { |a| a + 1 }
references::
    x in the following { |\x| x }

== Syntax

[source,bnf]
----
script ::= expression*

expression ::= let-expr
             | op-call

let-expr ::= 'let' let-name expression
let-name ::= symbol
           | bare-symbol
           | fn-call

fn-call ::= callable expression*

callable ::= bare-symbol
           | call-expr

call-expr ::= '&' op-call '/' UINTEGER

      op-call ::= op-expr
      op-expr ::= prec9-op-expr ( PREC9_OP prec9-op-expr )* <1>
prec9-op-expr ::= prec8-op-expr ( PREC8_OP prec8-op-expr )* <1>
prec8-op-expr ::= prec7-op-expr ( PREC7_OP prec7-op-expr )* <1>
prec7-op-expr ::= prec6-op-expr ( PREC6_OP prec6-op-expr )* <1>
prec6-op-expr ::= prec5-op-expr ( PREC5_OP prec5-op-expr )* <1>
prec5-op-expr ::= prec4-op-expr ( PREC4_OP prec4-op-expr )* <1>
prec4-op-expr ::= prec3-op-expr ( PREC3_OP prec3-op-expr )* <1>
prec3-op-expr ::= prec2-op-expr ( PREC2_OP prec2-op-expr )* <1>
prec2-op-expr ::= prec1-op-expr ( PREC1_OP prec1-op-expr )* <1>
prec1-op-expr ::= prec0-op-expr ( PREC0_OP prec0-op-expr )* <1>
prec0-op-expr ::= '(' expression ')'
                | fundamental-scalar
                | symbol
                | block
                | PREFIX_OP prec10-op-expr <2>
                | fn-call

fundamental-scalar ::= STRING
                     | INTEGER
                     | FLOAT

symbol ::= SYMBOL '/' UINTEGER
bare-symbol ::= SYMBOL

block ::= '{' [block-args] expression* '}'
        | '\' [block-args] expression
block-args ::= '|' block-arg* '|'
block-arg ::= bare-symbol
----
<1> `PREC<n>_OP` is dynamically updated with `<op>` upon parsing the appropriate `let op <n> <op>/2` expressions.
<2> `PREFIX_OP` is mostly the same as `PREC<n>_OP` but the source of updates is from prefix `let prefix <op>/1`.

Operators contain one or more of the following characters: `++-*/%&|@~!?^`<>=#.:+`
With the exception that `+#+` cannot stand by itself as that will be taken as a comment (and probably generate horrid errors during parsing).

let when/2 { |x chain| &&chain/1 x/0 }
let is/3 { |y case chain| { |x| if (x == y) case else { &chain/1 x } } }
let otherwise/2 { |y case| { |x| case } }

let map/2 { |self|f list|
    let tail cdr list
    if tail == nil {
        cons f car list
             nil
    }
    else {
        cons f car list
             self f tail
    }
